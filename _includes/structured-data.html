{%- comment -%} Site-wide JSON-LD (schema.org), included from footer.html on every page. One graph links the Person, the WebSite, the current page and its breadcrumb through stable @id URLs, so search engines and AI assistants resolve every page to the same entity. Facts come from _data/profile.yml and the collections. Pages can add their own nodes (the Brakeman visualizer adds WebApplication and FAQPage) and reference the person with {"@id": "/#person"}. {%- endcomment -%} {%- assign profile = site.data.profile -%} {%- assign home_url = '/' | absolute_url -%} {%- assign person_id = home_url | append: '#person' -%} {%- assign website_id = home_url | append: '#website' -%} {%- assign page_url = page.url | absolute_url -%} {%- assign page_id = page_url | append: '#webpage' -%} {%- assign is_home = false -%}{%- if page.url == '/' -%}{%- assign is_home = true -%}{%- endif -%} {%- comment -%} Only collection documents have an id; list pages only set "collection" in their front matter {%- endcomment -%} {%- assign is_document = false -%}{%- if page.id -%}{%- assign is_document = true -%}{%- endif -%} {%- assign page_type = 'WebPage' -%} {%- if is_home -%} {%- assign page_type = 'ProfilePage' -%} {%- elsif page.collection and is_document == false -%} {%- assign page_type = 'CollectionPage' -%} {%- endif -%} {%- assign page_name = page.title | default: site.title -%} {%- assign page_description = page.description | default: page.excerpt | default: site.description | strip_html | normalize_whitespace | truncate: 300 -%} {%- assign press_items = site.press | where_exp: 'item', 'item.draft != true' -%}
